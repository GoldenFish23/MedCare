
# User Table
Create table if not exists users(
    id int primary key,
    name text not null,
    password text not null,
    role text not null check(role in('admin', 'cashier')),
    created_at default datetime,
    deleted_at datetime,
);

+----+------+----------+------+------------+------------+
| id | name | password | role | created_at | deleted_at |
+----+------+----------+------+------------+------------+
+----+------+----------+------+------------+------------+
